   352→function ResultStep({
   353→  songTitle,
   354→  style,
   355→  songId,
   356→  coverUrl,
   357→  audioUrl,
   358→  lyrics,
   359→  duration,
   360→}: {
   361→  songTitle: string;
   362→  style: string;
   363→  songId: string;
   364→  coverUrl: string;
   365→  audioUrl: string;
   366→  lyrics: string;
   367→  duration: number;
   368→}) {
   369→  const router = useRouter();
   370→  const [isPlaying, setIsPlaying] = useState(false);
   371→  const audioRef = useRef<HTMLAudioElement | null>(null);
   372→  const [currentTime, setCurrentTime] = useState(0);
   373→
   374→  const formatTime = (s: number) => `${Math.floor(s / 60)}:${String(Math.floor(s % 60)).padStart(2, "0")}`;
   375→
   376→  useEffect(() => {
   377→    if (audioUrl && audioRef.current) {
   378→      audioRef.current.src = audioUrl;
   379→    }
   380→  }, [audioUrl]);
   381→
   382→  const togglePlay = () => {
   383→    if (!audioRef.current) return;
   384→    if (isPlaying) {
   385→      audioRef.current.pause();
   386→    } else {
   387→      audioRef.current.play().catch(() => toast.error("Audio non disponible"));
   388→    }
   389→    setIsPlaying(!isPlaying);
   390→  };
   391→
   392→  const handleDownload = () => {
   393→    if (audioUrl) {
   394→      const a = document.createElement("a");
   395→      a.href = audioUrl;
   396→      a.download = `${songTitle}.mp3`;
   397→      a.click();
   398→      toast.success("Téléchargement lancé !");
   399→    } else {
   400→      toast.info("Audio en préparation...");
   401→    }
   402→  };
   403→
   404→  const handleShare = () => {
   405→    const url = `${window.location.origin}/song/${songId}`;
   406→    navigator.clipboard?.writeText(url);
   407→    toast.success("Lien copié ! 🎵");
   408→  };
   409→
   410→  const durationMin = formatTime(duration);
   411→
   412→  return (
   413→    <div className="space-y-6">
   414→      <div className="text-center mb-4">
   415→        <motion.div
   416→          initial={{ scale: 0 }}
   417→          animate={{ scale: 1 }}
   418→          transition={{ type: "spring", duration: 0.6 }}
   419→          className="w-20 h-20 rounded-full bg-emerald-500/10 flex items-center justify-center mx-auto mb-4 glow-green"
   420→        >
   421→          <Check className="w-10 h-10 text-emerald-400" />
   422→        </motion.div>
   423→        <h2 className="text-3xl font-bold text-white">Ta chanson est prête ! 🎵</h2>
   424→        <p className="text-slate-400 mt-2">Écoute, télécharge ou partage ta création</p>
   425→      </div>
   426→
   427→      <Card className="glass p-6 max-w-lg mx-auto">
   428→        {/* Cover art */}
   429→        <div className="w-full aspect-square rounded-xl overflow-hidden mb-4 relative">
   430→          {coverUrl && !coverUrl.startsWith("/covers/") ? (
   431→            <img src={coverUrl} alt={songTitle} className="w-full h-full object-cover" />
   432→          ) : (
   433→            <div className="w-full h-full bg-gradient-to-br from-purple-500/30 via-pink-500/20 to-amber-500/20 flex items-center justify-center">
   434→              <Music className="w-20 h-20 text-white/40" />
   435→            </div>
   436→          )}
   437→          <div className="absolute inset-0 bg-gradient-to-t from-black/70 via-transparent to-transparent" />
   438→          <div className="absolute bottom-4 left-4 right-4">
   439→            <p className="text-white font-bold text-xl">{songTitle}</p>
   440→            <p className="text-white/70 text-sm">{style}</p>
   441→          </div>
   442→        </div>
   443→
   444→        {/* Real Audio Player */}
   445→        <div className="bg-white/5 rounded-lg p-4 flex items-center gap-3 mb-4">
   446→          <audio

   518→export function CreateFlowClient() {
   519→  const router = useRouter();
   520→  const { data: session } = useSession();
   521→  const [sidebarCollapsed, setSidebarCollapsed] = useState(false);
   522→  const [step, setStep] = useState(0);
   523→  const [style, setStyle] = useState("");
   524→  const [theme, setTheme] = useState("");
   525→  const [mood, setMood] = useState("");
   526→  const [title, setTitle] = useState("");
   527→  const [additionalPrompt, setAdditionalPrompt] = useState("");
   528→  const [generating, setGenerating] = useState(false);
   529→  const [progress, setProgress] = useState(0);
   530→  const [statusText, setStatusText] = useState("");
   531→  const [currentStage, setCurrentStage] = useState(0);
   532→  const [songId, setSongId] = useState<string | null>(null);
   533→  const [coverUrl, setCoverUrl] = useState("");
   534→  const [audioUrl, setAudioUrl] = useState("");
   535→  const [lyrics, setLyrics] = useState("");
   536→  const [duration, setDuration] = useState(180);
   537→  const [credits, setCredits] = useState<number | null>(null);
   538→  const abortRef = useRef(false);
   539→
   540→  const userId = (session?.user as any)?.id as string | undefined;
   541→  const userName = session?.user?.name || "Créateur";
   542→  const userPlan = (session?.user as any)?.plan || "decouverte";
   543→
   544→  // Fetch user credits
   545→  useEffect(() => {
   546→    if (userId) {
   547→      fetch(`/api/me/credits?userId=${userId}`)
   548→        .then((r) => r.json())
   549→        .then((data) => {
   550→          if (data.credits) setCredits(data.credits.credits);
   551→        })
   552→        .catch(() => {});
   553→    }
   554→  }, [userId]);
   555→
   556→  const canNext = () => {
   557→    switch (step) {
   558→      case 0: return !!style;
   559→      case 1: return !!theme;
   560→      case 2: return !!mood;
   561→      case 3: return !!title.trim();
   562→      default: return false;
   563→    }
   564→  };
   565→
   566→  const handleNext = () => {
   567→    if (step < 3) {
   568→      setStep(step + 1);
   569→    } else if (step === 3) {
   570→      if (!userId) {
   571→        toast.error("Tu dois être connecté pour générer une chanson");
   572→        return;
   573→      }
   574→
   575→      if (credits !== null && credits < 7) {
   576→        toast.error("Crédits insuffisants. Passe à un plan supérieur.");
   577→        router.push("/subscription");
   578→        return;
   579→      }
   580→
   581→      // Start generation
   582→      setStep(4);
   583→      setGenerating(true);
   584→      setProgress(0);
   585→      setStatusText("Analyse de ta demande...");
   586→      setCurrentStage(0);
   587→      abortRef.current = false;
   588→
   589→      // Show generation stages progressively
   590→      let elapsed = 0;
   591→      generationStages.forEach((stage, i) => {
   592→        elapsed += stage.duration;
   593→        setTimeout(() => {
   594→          if (!abortRef.current) {
   595→            setCurrentStage(i);
   596→            setProgress(stage.progress);
   597→            setStatusText(`${stage.name} en cours...`);
   598→          }
   599→        }, elapsed - stage.duration);
   600→      });
   601→
   602→      // Call the real AI generation API
   603→      fetch("/api/generate", {
   604→        method: "POST",
   605→        headers: { "Content-Type": "application/json" },
   606→        body: JSON.stringify({
   607→          userId,
   608→          style,
   609→          theme,
   610→          mood,
   611→          title,
   612→          language: "fr",
   613→          additionalPrompt,

   621→

   623→            toast.error(data.error);

   625→            setGenerating(false);
   626→            return;
   627→          }

   629→          // Complete all stages

   631→          setProgress(100);

   634→          setTimeout(() => {
   635→            if (abortRef.current) return;
   636→            setGenerating(false);

   638→            setCoverUrl(data.song.coverUrl);
   639→            setAudioUrl(data.song.audioUrl);
   640→            setLyrics(data.song.lyrics || "");
   641→            setDuration(data.song.duration || 180);

   671→        <Header title="Studio de création" userName={userName} userPlan={userPlan} />
   672→
   673→        <div className="p-6 max-w-5xl mx-auto">
